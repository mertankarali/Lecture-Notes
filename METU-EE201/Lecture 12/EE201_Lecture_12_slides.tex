\begin{frame}{Review: node voltage analysis}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{The procedure}
        \begin{itemize}\itemsep3pt
          \item Choose the ground; label the node voltages $\Vcol{\mathbf{e}}$.
          \item KVL holds automatically: $\Vcol{\mathbf{v}} = A^T\Vcol{\mathbf{e}}$.
          \item Write KCL at every node except the ground.
          \item Matrix form: $Y_n\Vcol{\mathbf{e}} = \Icol{\mathbf{i}_n}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \onslide<2->{%
      \begin{block}{Voltage sources}
        \begin{itemize}\itemsep3pt
          \item From the ground to node $k$: $\e{k}$ is known.
          \item Between two nodes: a supernode.
          \item A dependent source: one more constraint.
        \end{itemize}
      \end{block}}
    \end{column}
  \end{columns}
  \vfill
  \onslide<3->{%
  \begin{center}
    Node voltages make \alert{KVL} automatic, and we write \alert{KCL}.\\[4pt]
    \onslide<4->{Can we do the opposite?}
  \end{center}}
  \vfill
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Motivation}
  \vspace{2pt}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{block}{Node voltages}
        \small
        \begin{itemize}\itemsep4pt
          \item Unknowns: the $n - 1$ node voltages $\Vcol{\mathbf{e}}$.
          \item KVL holds automatically: $\Vcol{\mathbf{v}} = A^T\Vcol{\mathbf{e}}$.
          \item We write KCL: $n - 1$ equations.
        \end{itemize}
      \end{block}

\begin{frame}{Example 12.1 in matrix form}

    $Z_M$ is the \alert{mesh impedance matrix}: \ $Z_M\Icol{\mathbf{i}_M} = \Vcol{\mathbf{v}_M}$.\\[3pt]

\begin{frame}{$Z_M$ from the graph: KVL, KCL and Ohm's law in blocks}
  \small
  Number the $b_R$ resistors first, then the $b_S$ voltage sources:
  \[
    M = [\,M_R \ \ M_S\,], \qquad
    \Icol{\mathbf{i}} = \begin{bmatrix} \Icol{\mathbf{i}_R}\\ \Icol{\mathbf{i}_S} \end{bmatrix}, \qquad
    \Vcol{\mathbf{v}} = \begin{bmatrix} \Vcol{\mathbf{v}_R}\\ \Vcol{\mathbf{v}_S} \end{bmatrix}.
  \]
  \vfill
  \onslide<2->{%
  \textbf{KVL:} \
  $M\Vcol{\mathbf{v}} = [\,M_R \ \ M_S\,]\begin{bmatrix} \Vcol{\mathbf{v}_R}\\ \Vcol{\mathbf{v}_S} \end{bmatrix}
   = M_R\Vcol{\mathbf{v}_R} + M_S\Vcol{\mathbf{v}_S} = \mathbf{0}$}
  \vfill
  \onslide<3->{%
  \textbf{KCL:} \ $M^T$ splits into \alert{rows}:
  \[
    \Icol{\mathbf{i}} = M^T\Icol{\mathbf{i}_M}: \qquad
    \begin{bmatrix}
      \hrow{metugray!15}{\Icol{\mathbf{i}_R}}\\
      \hrow{cvolt!12}{\Icol{\mathbf{i}_S}}
    \end{bmatrix}
    =
    \begin{bmatrix}
      \hrow{metugray!15}{M_R^T}\\
      \hrow{cvolt!12}{M_S^T}
    \end{bmatrix}
    \Icol{\mathbf{i}_M}
    \qquad
    \begin{array}{l}
      \textcolor{metugray}{b_R \text{ resistor rows: } \Icol{\mathbf{i}_R} = M_R^T\Icol{\mathbf{i}_M}}\\[4pt]
      \textcolor{cvolt}{b_S \text{ source rows: } \Icol{\mathbf{i}_S} = M_S^T\Icol{\mathbf{i}_M}}
    \end{array}
  \]}
  \vfill
  \onslide<4->{\centering Only the resistor rows are needed: the source currents follow afterwards.\par}
  \vfill
  \onslide<5->{%
  \textbf{Resistors:} \ $\Vcol{\mathbf{v}_R} = R\,\Icol{\mathbf{i}_R}$, \
  $R = \mathrm{diag}(R_1, \dots, R_{b_R})$, one resistance per resistor.}
  \vfill
\end{frame}

\begin{frame}{$Z_M$ from the graph: the general result}
  \vfill
  Put Ohm's law and the resistor rows of KCL into KVL:
  \[
    M_R\Vcol{\mathbf{v}_R} + M_S\Vcol{\mathbf{v}_S}
    = M_R\, R\, \underbrace{M_R^T\Icol{\mathbf{i}_M}}_{\Icol{\mathbf{i}_R}} + M_S\Vcol{\mathbf{v}_S} = \mathbf{0}
    \quad\Rightarrow\quad
    M_R\, R\, M_R^T\,\Icol{\mathbf{i}_M} = -M_S\Vcol{\mathbf{v}_S}
  \]
  \vfill
  \onslide<2->{%
  \begin{center}
    \begin{tikzpicture}
      \node[draw=metured, thick, fill=metured!5, rounded corners=3pt, inner sep=8pt]
        {$Z_M = M_R\, R\, M_R^T, \qquad \Vcol{\mathbf{v}_M} = -M_S\Vcol{\mathbf{v}_S}$};
    \end{tikzpicture}
  \end{center}}
  \vfill
  \onslide<3->{%
  \begin{center}
    $Z_M^T = M_R\, R^T M_R^T = M_R\, R\, M_R^T = Z_M$, \ since $R$ is diagonal.\\[4pt]
    So $Z_M$ is \alert{always symmetric} for resistors and voltage sources.
  \end{center}}
  \vfill
\end{frame}

\begin{frame}{Method I: the voltage as an unknown}

      The constraint: $\Icol{i_s}$ carries $\im{1}$ down and $\im{2}$ up.\\[4pt]}
      \onslide<6->{%
      Three unknowns, three equations; the last row is the constraint.\\[4pt]}
      \onslide<7->{%
      It works, but the supermesh avoids the extra unknown.}
    \end{column}
  \end{columns}
\end{frame}

  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \textbf{Question.} Can we get rid of the unknown $\Vcol{v}$?\\[8pt]
      \onslide<2->{%
      Add the KVL equations of meshes 1 and 2.\\[3pt]
      $\Vcol{v}$ enters mesh 1 with $+$ and mesh 2 with $-$, so it \alert{cancels}.}
    \end{column}

\begin{frame}{Dependent sources}
  \vfill
  \begin{itemize}\itemsep10pt
    \item A dependent source is handled like an independent one of the same kind.
    \item<2-> Its control variable ($\Vcol{v_x}$ here) is written with mesh currents: this
      gives the \alert{constraint equation}.
    \item<3-> A dependent \alert{voltage} source is the easy case: it adds a term to KVL,
      just as a dependent current source added a term to KCL previously.
  \end{itemize}
  \vfill
  \onslide<4->{%
  \begin{block}{Remark}
    If a dependent current source is on mesh $k$ only, do \alert{not} write KVL around
    mesh $k$: its voltage is not known.\\[4pt]
    Write the constraint that the dependent source gives instead.
  \end{block}}
  \vfill
\end{frame}

\begin{frame}{Node or mesh?}
  \vfill
  \textbf{Question.} For Example 12.4, which method needs fewer equations?
  \vfill
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \onslide<2->{%
      \begin{block}{Mesh currents}
        \small
        $\ell = 3$ unknowns.\\[3pt]
        Each current source fixes one of them: $3 - 2 = \alert{1}$ KVL equation is left.
      \end{block}}
    \end{column}
    \begin{column}{0.485\textwidth}
      \onslide<3->{%
      \begin{block}{Node voltages}
        \small
        $n - 1 = 3$ unknowns.\\[3pt]
        Each voltage source fixes one of them: $3 - 1 = \alert{2}$ KCL equations are left.
      \end{block}}
    \end{column}
  \end{columns}
  \vfill
  \onslide<4->{%
  \begin{center}
    \begin{tikzpicture}
      \node[draw=metured, thick, fill=metured!5, rounded corners=3pt, inner sep=7pt,
            text width=0.82\textwidth, align=center]
        {\textbf{Tip.} Count before you start: mesh, $\ell$ minus the current sources;
         node, $n - 1$ minus the voltage sources.\\[2pt]
         Mesh current analysis needs a planar circuit; node voltage analysis always works.};
    \end{tikzpicture}
  \end{center}}
