\begin{frame}{$Y_n$ from the graph: KCL, KVL and Ohm's law in blocks}
  Resistors first, then current sources:
  \[
    A = [\,A_R \ \ A_S\,], \qquad
    \Icol{\mathbf{i}} = \begin{bmatrix} \Icol{\mathbf{i}_R}\\ \Icol{\mathbf{i}_S} \end{bmatrix}, \qquad
    \Vcol{\mathbf{v}} = \begin{bmatrix} \Vcol{\mathbf{v}_R}\\ \Vcol{\mathbf{v}_S} \end{bmatrix}
  \]
  \vfill
  \onslide<2->{%
  \textbf{KCL:} \
  $A\Icol{\mathbf{i}} = [\,A_R \ \ A_S\,]\begin{bmatrix} \Icol{\mathbf{i}_R}\\ \Icol{\mathbf{i}_S} \end{bmatrix}
   = A_R\Icol{\mathbf{i}_R} + A_S\Icol{\mathbf{i}_S} = \mathbf{0}$}
  \vfill
  \onslide<3->{%
  \textbf{KVL:}
  \[
    \begin{bmatrix}
      \hrow{metugray!15}{\Vcol{\mathbf{v}_R}}\\
      \hrow{ccur!12}{\Vcol{\mathbf{v}_S}}
    \end{bmatrix}
    =
    \begin{bmatrix}
      \hrow{metugray!15}{A_R^T}\\
      \hrow{ccur!12}{A_S^T}
    \end{bmatrix}\Vcol{\mathbf{e}}
    \qquad
    \onslide<4->{\Rightarrow\qquad
    \Vcol{\mathbf{v}_R} = A_R^T\Vcol{\mathbf{e}} \ \ \text{(resistor rows only)}}
  \]}
  \vfill
  \onslide<5->{%
  \textbf{Ohm:} \ $\Icol{\mathbf{i}_R} = G\,\Vcol{\mathbf{v}_R}$, \quad
  $G = \mathrm{diag}(G_1, \dots, G_{b_R})$}
  \vfill
\end{frame}

%--------------------------------------------------------------
\begin{frame}{$Y_n$ from the graph: the general result}
  \vfill
  Ohm and KVL into KCL:
  \[
    A_R\Icol{\mathbf{i}_R} + A_S\Icol{\mathbf{i}_S}
    = A_R\, G\, \underbrace{A_R^T\Vcol{\mathbf{e}}}_{\Vcol{\mathbf{v}_R}} + A_S\Icol{\mathbf{i}_S} = \mathbf{0}
    \quad\Rightarrow\quad
    A_R\, G\, A_R^T\,\Vcol{\mathbf{e}} = -A_S\Icol{\mathbf{i}_S}
  \]
  \vfill
  \onslide<2->{%
  \begin{center}
    \begin{tikzpicture}
      \node[draw=metured, fill=metured!6, rounded corners=3pt, inner sep=6pt]
        {$Y_n = A_R\, G\, A_R^T, \qquad \Icol{\mathbf{i}_n} = -A_S\Icol{\mathbf{i}_S}$};
    \end{tikzpicture}
  \end{center}}
  \vfill
  \onslide<3->{%
  \begin{center}
    $Y_n^T = A_R\, G^T A_R^T = Y_n$: \ \alert{always symmetric}
  \end{center}}
  \vfill
\end{frame}
