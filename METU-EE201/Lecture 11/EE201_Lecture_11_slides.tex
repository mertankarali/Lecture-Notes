\begin{frame}{Review: two ways to solve an LTI resistive circuit}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Transform the circuit}
        \begin{itemize}\itemsep3pt
          \item Substitution, equivalent one-ports.
          \item Series and parallel connections.
          \item Source transformation.
        \end{itemize}
        \vspace{3pt}
        Works for \alert{many} circuits, but not all.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Keep the original circuit}
        \begin{itemize}\itemsep3pt
          \item KCL: $A\Icol{\mathbf{i}} = \mathbf{0}$.
          \item KVL: $\Vcol{\mathbf{v}} = A^T\Vcol{\mathbf{e}}$.
          \item One element law per branch.
        \end{itemize}
        \vspace{3pt}
        Works for \alert{every} circuit.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    With the node voltages $\Vcol{\mathbf{e}}$ as the unknowns, KVL holds
    \alert{automatically}.\\[4pt]
    Only KCL is left to write.
  \end{center}
  \vfill
\end{frame}

          \item<2-> Simplifying \alert{fundamentally changes} the circuit; that is its whole

          \item<3-> To find the quantities of the original circuit, we must go back step by step.

\begin{frame}{$Y_n$ from the graph: KCL, KVL and Ohm's law in blocks}
  \small
  Number the $b_R$ resistors first, then the $b_S$ current sources:
  \[
    A = [\,A_R \ \ A_S\,], \qquad
    \Icol{\mathbf{i}} = \begin{bmatrix} \Icol{\mathbf{i}_R}\\ \Icol{\mathbf{i}_S} \end{bmatrix}, \qquad
    \Vcol{\mathbf{v}} = \begin{bmatrix} \Vcol{\mathbf{v}_R}\\ \Vcol{\mathbf{v}_S} \end{bmatrix}.
  \]
  \vfill
  \onslide<2->{%
  \textbf{KCL:} \
  $A\Icol{\mathbf{i}} = [\,A_R \ \ A_S\,]\begin{bmatrix} \Icol{\mathbf{i}_R}\\ \Icol{\mathbf{i}_S} \end{bmatrix}
   = A_R\Icol{\mathbf{i}_R} + A_S\Icol{\mathbf{i}_S} = \mathbf{0}$}
  \vfill
  \onslide<3->{%
  \textbf{KVL:} \ $A^T$ splits into \alert{rows}:
  \[
    \Vcol{\mathbf{v}} = A^T\Vcol{\mathbf{e}}:
    \qquad
    \left[\begin{array}{c}
      \rowcolor{metugray!15} \Vcol{\mathbf{v}_R}\\
      \rowcolor{ccur!12} \Vcol{\mathbf{v}_S}
    \end{array}\right]
    =
    \left[\begin{array}{c}
      \rowcolor{metugray!15} A_R^T\\
      \rowcolor{ccur!12} A_S^T
    \end{array}\right]\Vcol{\mathbf{e}}
    \qquad
    \begin{array}{l}
      \textcolor{metugray}{b_R \text{ resistor rows:}} \ \ \Vcol{\mathbf{v}_R} = A_R^T\Vcol{\mathbf{e}}\\[2pt]
      \textcolor{ccur}{b_S \text{ source rows:}} \ \ \Vcol{\mathbf{v}_S} = A_S^T\Vcol{\mathbf{e}}
    \end{array}
  \]}
  \onslide<4->{%
  \begin{center}
    Only the \alert{resistor rows} are needed: the source voltages follow afterwards.
  \end{center}}
  \vfill
  \onslide<5->{%
  \textbf{Resistors:} \ $\Icol{\mathbf{i}_R} = G\,\Vcol{\mathbf{v}_R}$, \
  $G = \mathrm{diag}(G_1, \dots, G_{b_R})$, one conductance per resistor.}
  \vfill
\end{frame}

\begin{frame}{$Y_n$ from the graph: the general result}
  \vfill
  Put Ohm's law and the resistor rows of KVL into KCL:
  \[
    A_R\Icol{\mathbf{i}_R} + A_S\Icol{\mathbf{i}_S}
    = A_R\, G\, \underbrace{A_R^T\Vcol{\mathbf{e}}}_{\Vcol{\mathbf{v}_R}} + A_S\Icol{\mathbf{i}_S} = \mathbf{0}
    \quad\Rightarrow\quad
    A_R\, G\, A_R^T\,\Vcol{\mathbf{e}} = -A_S\Icol{\mathbf{i}_S}
  \]
  \vfill
  \onslide<2->{%
  \begin{center}
    \begin{tikzpicture}
      \node[draw=metured, fill=metured!6, rounded corners=3pt, inner sep=6pt]
        {$Y_n = A_R\, G\, A_R^T, \qquad \Icol{\mathbf{i}_s} = -A_S\Icol{\mathbf{i}_S}$};
    \end{tikzpicture}
  \end{center}}
  \vfill
  \onslide<3->{%
  \begin{center}
    $Y_n^T = A_R\, G^T A_R^T = A_R\, G\, A_R^T = Y_n$, \ since $G$ is diagonal.\\[4pt]
    So $Y_n$ is \alert{always symmetric} for resistors and current sources.
  \end{center}}
  \vfill
\end{frame}

\begin{frame}{Dependent sources}
  \vfill
  \begin{itemize}\itemsep10pt
    \item A dependent source is handled like an independent one of the same kind.
    \item<2-> Its control variable ($\Vcol{v_x}$ here) is written with node voltages: this
      gives the \alert{constraint equation}.
  \end{itemize}
  \vfill
  \onslide<3->{%
  \begin{block}{Remark}
    If a dependent voltage source sits between node $k$ and the ground, do
    \alert{not} write KCL at node $k$: its current is not known.\\[4pt]
    Write the constraint that the dependent source gives instead.
  \end{block}}
  \vfill
\end{frame}
