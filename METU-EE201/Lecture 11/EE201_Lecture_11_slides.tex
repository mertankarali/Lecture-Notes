\begin{frame}{Review: two ways to solve an LTI resistive circuit}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Transform the circuit}
        \begin{itemize}\itemsep3pt
          \item Substitution, equivalent one-ports.
          \item Series and parallel connections.
          \item Source transformation.
        \end{itemize}
        \vspace{3pt}
        Works for \alert{many} circuits, but not all.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Keep the original circuit}
        \begin{itemize}\itemsep3pt
          \item KCL: $A\Icol{\mathbf{i}} = \mathbf{0}$.
          \item KVL: $\Vcol{\mathbf{v}} = A^T\Vcol{\mathbf{e}}$.
          \item One element law per branch.
        \end{itemize}
        \vspace{3pt}
        Works for \alert{every} circuit.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    With the node voltages $\Vcol{\mathbf{e}}$ as the unknowns, KVL holds
    \alert{automatically}.\\[4pt]
    Only KCL is left to write.
  \end{center}
  \vfill
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Motivation}
  \vspace{2pt}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{block}{Simplifying the circuit}
        \small
        \begin{itemize}\itemsep4pt
          \item So far, we solved circuits by simplifying them.
          \item<2-> Simplifying \alert{fundamentally changes} the circuit; that is its whole
            point.
          \item<3-> To find the quantities of the original circuit, we must go back step by step.
          \item<4-> Some circuits cannot be simplified at all.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.485\textwidth}
      \onslide<5->{%
      \begin{block}{Node voltage analysis}
        \small
        \begin{itemize}\itemsep4pt
          \item It works on the \alert{original} circuit, without changing it.
          \item<6-> Its unknowns are only the $n - 1$ \alert{node voltages}.
          \item<7-> From them, \alert{every} branch voltage and current follows.
          \item<8-> It works for every LTI resistive circuit; voltage sources need small additions, such as the supernode.
        \end{itemize}
      \end{block}}
    \end{column}
  \end{columns}
  \vfill
  \onslide<9->{\centering Later, mesh current analysis will do the same with mesh currents.\par}
  \vfill
\end{frame}

\begin{frame}{Example 11.1 in matrix form}
  \vfill
  With the conductances $G_k = 1/R_k$:
  \[
    \begin{aligned}
      (G_1 + G_2 + G_5)\,\e{1} - G_2\,\e{2} - G_5\,\e{3} &= \Icol{i_{s1}}\\
      -G_2\,\e{1} + (G_2 + G_3 + G_4)\,\e{2} - G_4\,\e{3} &= 0\\
      -G_5\,\e{1} - G_4\,\e{2} + (G_4 + G_5)\,\e{3} &= -\Icol{i_{s2}}
    \end{aligned}
  \]
  \vfill
  \onslide<2->{%
  \[
    \underbrace{\begin{bmatrix}
      G_1 + G_2 + G_5 & -G_2 & -G_5\\
      -G_2 & G_2 + G_3 + G_4 & -G_4\\
      -G_5 & -G_4 & G_4 + G_5
    \end{bmatrix}}_{Y_n}
    \underbrace{\begin{bmatrix} \e{1}\\ \e{2}\\ \e{3} \end{bmatrix}}_{\Vcol{\mathbf{e}}}
    =
    \underbrace{\begin{bmatrix} \Icol{i_{s1}}\\ 0\\ -\Icol{i_{s2}} \end{bmatrix}}_{\Icol{\mathbf{i}_s}}
  \]}
  \vfill
  \onslide<3->{\begin{center}
    $Y_n$ is the \alert{node admittance matrix}: \ $Y_n\Vcol{\mathbf{e}} = \Icol{\mathbf{i}_s}$.
  \end{center}}
  \vfill
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Writing $Y_n$ directly from the circuit}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.8]
        \exA{0}
      \end{tikzpicture}\\[8pt]
      {\small
      $\begin{bmatrix}
        \alert{G_1 + G_2 + G_5} & -G_2 & -G_5\\
        -G_2 & \alert{G_2 + G_3 + G_4} & -G_4\\
        -G_5 & -G_4 & \alert{G_4 + G_5}
      \end{bmatrix}$}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      \textbf{Question.} Can you read each entry of $Y_n$ directly from the circuit?\\[10pt]
      \onslide<2->{%
      \textbf{Diagonal} $(k,k)$: the sum of the conductances \alert{connected to} node $k$.\\[7pt]}
      \onslide<3->{%
      \textbf{Off-diagonal} $(k,j)$: minus the sum of the conductances \alert{between} nodes $k$
      and $j$.\\[7pt]}
      \onslide<4->{%
      \textbf{Right side} $k$: the sum of the source currents \alert{entering} node $k$.\\[7pt]}
      \onslide<5->{%
      So $Y_n$ is \alert{symmetric}: the entry $(k,j)$ equals the entry $(j,k)$.}
    \end{column}
  \end{columns}
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Why is $Y_n$ symmetric?}
  \begin{block}{The reduced incidence matrix $A$}
    \small
    One row for each node except the ground, one column for each branch.\\[2pt]
    Entry $+1$ if the branch leaves the node, $-1$ if it enters, $0$ otherwise.
  \end{block}
  \vspace{6pt}
  \onslide<2->{%
  Number the resistors first, then the current sources:
  \ $A = [\,A_R \ \ A_S\,]$, \ $\Icol{\mathbf{i}} = \begin{bmatrix} \Icol{\mathbf{i}_R}\\ \Icol{\mathbf{i}_S} \end{bmatrix}$.}
  \vfill
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\itemsep6pt
        \item<3-> KCL: \ $A_R\Icol{\mathbf{i}_R} + A_S\Icol{\mathbf{i}_S} = \mathbf{0}$.
        \item<4-> Resistors: \ $\Icol{\mathbf{i}_R} = G\,\Vcol{\mathbf{v}_R}$, \
          $G = \mathrm{diag}(G_1, \dots, G_5)$.
        \item<5-> KVL: \ $\Vcol{\mathbf{v}_R} = A_R^T\Vcol{\mathbf{e}}$.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \onslide<6->{%
      \[
        \underbrace{A_R\, G\, A_R^T}_{Y_n}\,\Vcol{\mathbf{e}} = \underbrace{-A_S\Icol{\mathbf{i}_S}}_{\Icol{\mathbf{i}_s}}
      \]}
      \onslide<7->{%
      \[
        Y_n^T = A_R\, G^T A_R^T = A_R\, G\, A_R^T = Y_n
      \]}
    \end{column}
  \end{columns}
  \vfill
  \onslide<8->{\begin{center}
    $G$ is diagonal, so $Y_n$ is symmetric.\\[3pt]
    $-A_S\Icol{\mathbf{i}_S}$ adds up the source currents entering each node.
  \end{center}}
  \vfill
\end{frame}

    \begin{column}{0.54\textwidth}
      \small
      \textbf{Question.} Redo the example with node 3 as the ground. Does the answer
      change?\\[8pt]
      \onslide<2->{%
      Now $\Vcol{v_x}$ is between node 2 and the ground: no supernode is needed.\\[6pt]}
      \onslide<3->{%
      Unknowns: $\ep{1}$, $\ep{2}$ and $\ep{0}$, the old ground.\\[8pt]}
      \onslide<4->{%
      \textbf{Node 1:} \
      $\displaystyle -5 + \frac{\ep{1}}{2} + \frac{\ep{1} - \ep{0}}{4} + \frac{\ep{1} - \ep{2}}{2} = 0$\\[5pt]
      \textbf{Node 0:} \
      $\displaystyle 5 + \frac{\ep{0} - \ep{1}}{4} + \frac{\ep{0} - \ep{2}}{3} = 0$\\[5pt]}
      \onslide<5->{%
      \textbf{Constraint:} \ $\ep{2} = \Vcol{v_x} = \ep{1} - \ep{0}$; \ no KCL at node 2.}
    \end{column}
  \end{columns}
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Exercise: the solution}
  \vfill
  \begin{itemize}\itemsep8pt
    \item Put the constraint into both KCL equations:
      \[
        3\ep{1} + \ep{0} = 20, \qquad -7\ep{1} + 11\ep{0} = -60 .
      \]
    \item<2-> So $\ep{1} = 7\V$, \ $\ep{0} = -1\V$, \ $\ep{2} = 8\V$.
    \item<3-> The $3\ohm$ is between nodes 2 and 0:
      $\Pcol{P_{3\Omega}} = \dfrac{(\ep{2} - \ep{0})^2}{3} = \dfrac{9^2}{3} = \Pcol{27\W}$,
      \alert{the same}.
  \end{itemize}
  \vfill
  \onslide<4->{%
  \begin{center}
    Every node voltage drops by the same $\e{3} = 1\V$: \ $8 \to 7$, \ $9 \to 8$, \ $0 \to -1$.\\[5pt]
    The branch voltages, so the currents and the powers, do not change.
  \end{center}}
  \vfill
  \onslide<5->{%
  \begin{center}
    \alert{Tip:} put the ground at one end of a voltage source; this saves a supernode.
  \end{center}}
  \vfill
\end{frame}
