\textbf{Motivation.} Previously, we simplified circuits with equivalent one-ports: series and
parallel connections and source transformation. A one-port with dependent sources is not a
series--parallel connection of resistors. The first part finds its one equivalent resistor: the
\textbf{input resistance}. The second part shows that an LTI resistive circuit is
\textbf{linear} in its independent sources; \textbf{superposition} then solves a circuit with
several sources one source at a time. The last part adds independent sources inside the
one-port: the \textbf{Th\'evenin and Norton} equivalents. A take-home part (appendix) gives the
\textbf{$\boldsymbol{\Delta}$--Y transform} for three resistors that are neither in series nor
in parallel.

%==============================================================
\section{The input resistance of a one-port}

\begin{definitionbox}{Input resistance}
\begin{minipage}[c]{0.5\linewidth}
Let a one-port $N$ contain only LTI resistors, LTI dependent sources and ideal transformers,
with every control variable inside $N$. Then $N$ is equivalent to a single LTI resistor
$\Rin$, the \textbf{input resistance} of $N$: \ $\Vcol{v} = \Rin\,\Icol{i}$ at its port.
\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}
\centering
\begin{tikzpicture}[scale=0.62]
  \oneport[4.6]{\footnotesize only LTI resistors,\\[-2pt] \footnotesize dependent sources,\\[-2pt]
    \footnotesize ideal transformers}
  \node[font=\large] at (6.5,1) {$\equiv$};
  \begin{scope}[shift={(9.1,0)}]
    \rinres
  \end{scope}
\end{tikzpicture}
\end{minipage}
\end{definitionbox}

\begin{minipage}[c]{0.5\linewidth}
With dependent sources, series and parallel rules are not enough to find $\Rin$. A
\textbf{test source} finds it. Connect a $1\A$ current source to the port of $N$ (circuit I)
and to $\Rin$ (circuit II). Since $N \equiv \Rin$, the port voltages are equal:
$\Vcol{v_{\mathrm{I}}} = \Vcol{v_{\mathrm{II}}} = \Rin \cdot 1\A$, so
\[
  \Rin = \frac{\Vcol{v_{\mathrm{I}}}}{1\A}: \quad \text{solve circuit (I)}.
\]
\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}
\centering
\begin{tikzpicture}[scale=0.62]
  \testcfg{0} \node at (1.45,-0.45) {(I)};
  \begin{scope}[shift={(5.9,0)}]
    \testcfg{1} \node at (1.45,-0.45) {(II)};
  \end{scope}
\end{tikzpicture}
\end{minipage}

\medskip
\begin{examplebox}[Example 13.1 --- an input resistance]
\begin{minipage}[c]{0.46\linewidth}
\centering
\begin{tikzpicture}[scale=0.72]
  \exR{0}
\end{tikzpicture}\\[6pt]
\begin{tikzpicture}[scale=0.72]
  \exR{1}
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.5\linewidth}
\textbf{Question.} Find $\Rin$.

\textbf{Answer.} Connect a $1\A$ test source and take the bottom node as the ground. KCL at
node 1:
\[
  \frac{\e{1} - \frac32\,\Icol{i_x}}{3} + \frac{\e{1}}{2} - 1 = 0 .
\]
The test current enters at a and leaves at b, so $\Icol{i_x} = -1\A$. First plug it in:
\[
  \frac{\e{1} + \frac32}{3} + \frac{\e{1}}{2} - 1 = 0
  \ \Rightarrow\ \tfrac56\,\e{1} = \tfrac12
  \ \Rightarrow\ \e{1} = \tfrac35\V .
\]
So $\Vcol{v_{\mathrm{I}}} = \e{1} = \frac35\V$ and $\Rin = \frac35\ohm$.
\end{minipage}

\smallskip
With the dependent source replaced by a short circuit, $\Rin$ would be $3 \parallel 2 = \frac65\ohm$:
the dependent source changes the input resistance.
\end{examplebox}

\section{Linearity}

In an LTI resistive circuit, the independent sources are the \textbf{inputs}. Collect the
voltages of the independent voltage sources and the currents of the independent current
sources in the \textbf{input vector} $\mathbf{u} = [\,u_1 \ \cdots \ u_m\,]^T$. An
\textbf{output} $y$ is any branch voltage or current.

\begin{definitionbox}{Linearity}
\begin{minipage}[c]{0.56\linewidth}
\textbf{Input $u_j$ alone} (the other inputs zero): the output is
\[
  y_j = k_j\,u_j ,
\]
with $k_j$ a constant set by the resistors and the dependent-source gains.

\textbf{All inputs:} the output is the sum of these parts,
\[
  y = y_1 + y_2 + \cdots + y_m = k_1u_1 + \cdots + k_mu_m = \mathbf{k}\,\mathbf{u},
\]
with the row vector $\mathbf{k} = [\,k_1 \ \cdots \ k_m\,]$.
\end{minipage}\hfill
\begin{minipage}[c]{0.4\linewidth}
\centering
\begin{tikzpicture}[scale=0.85]
  \blockdiag
\end{tikzpicture}
\end{minipage}
\end{definitionbox}

\begin{examplebox}[Example 13.2 --- the row vector $\mathbf{k}$]
\begin{minipage}[c]{0.44\linewidth}
\centering
\begin{tikzpicture}[scale=0.72]
  \exL
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.54\linewidth}
\textbf{Question.} The inputs are $\mathbf{u} = [\,\Vcol{v_s} \ \ \Icol{i_s}\,]^T$. For the
output $y = \Icol{i}$, find $k_1$, $k_2$ with $\Icol{i} = \Icol{i_1} + \Icol{i_2}$,
$\Icol{i_1} = k_1\Vcol{v_s}$, $\Icol{i_2} = k_2\Icol{i_s}$.

\textbf{Answer.} KVL around the left mesh, with $\Icol{i} + \Icol{i_s}$ down through $R_2$:
\[
  -\Vcol{v_s} + R_1\Icol{i} + R_2(\Icol{i} + \Icol{i_s}) = 0 ,
\]
\[
  \Icol{i} = \underbrace{\frac{1}{R_1 + R_2}\,\Vcol{v_s}}_{\Icol{i_1}}
  \ \underbrace{-\,\frac{R_2}{R_1 + R_2}\,\Icol{i_s}}_{\Icol{i_2}} ,
\]
so $k_1 = 1/(R_1 + R_2)$ and $k_2 = -R_2/(R_1 + R_2)$.
\end{minipage}
\end{examplebox}

\begin{ideabox}[The implication of linearity]
For input vectors $\mathbf{u}$, $\mathbf{w}$ and numbers $\alpha$, $\beta$:
\[
  y(\alpha\,\mathbf{u} + \beta\,\mathbf{w}) = \alpha\,y(\mathbf{u}) + \beta\,y(\mathbf{w}) .
\]
\end{ideabox}

\section{Superposition}

Split the input vector into one source at a time. With two sources,
\[
  y\!\left(\begin{bmatrix} \Vcol{v_s}\\ \Icol{i_s} \end{bmatrix}\right)
  = y\!\left(\begin{bmatrix} \Vcol{v_s}\\ 0 \end{bmatrix}\right)
  + y\!\left(\begin{bmatrix} 0\\ \Icol{i_s} \end{bmatrix}\right) :
\]
the output is the sum of the outputs with one independent source at a time. The other
sources are \textbf{killed}.

\begin{definitionbox}{Killing a source}
Killing an independent voltage source ($\Vcol{v_s} = 0$) replaces it with a \textbf{short
circuit}. Killing an independent current source ($\Icol{i_s} = 0$) replaces it with an
\textbf{open circuit}. Do \textbf{not} kill dependent sources: they are not inputs.
\end{definitionbox}

$\Icol{i_2}$ with the $20\V$ source killed.

\[
  -20 + 15\,\Icol{i_x} + 10\,\Icol{i_1} - 5\,\Icol{i_x} = 0
\]
\[
  \Rightarrow\ 20\,\Icol{i_1} = 20 \ \Rightarrow\ \Icol{i_1} = 1\A .
\]
\end{minipage}\hfill

\]

\textbf{Answer.} By linearity, $\Vcol{v} = \mathbf{k}\,[\,\Vcol{v_s} \ \ \Icol{i_s}\,]^T$
for some $\mathbf{k}$. Write the third input with the first two:
\[
  \begin{bmatrix} 4\\ 4 \end{bmatrix}
  = \alpha \begin{bmatrix} 12\\ 0 \end{bmatrix} + \beta \begin{bmatrix} -6\\ 6 \end{bmatrix}
  \ \Rightarrow\ 6\beta = 4, \ \ 12\alpha - 6\beta = 4
  \ \Rightarrow\ \alpha = \beta = \tfrac23 .
\]
Then $\Vcol{v_3} = \alpha\,\Vcol{v_1} + \beta\,\Vcol{v_2} = \frac23\,\Vcol{v_1} + \frac23\,\Vcol{v_2}$.
Equivalently, $\mathbf{k} = [\,\Vcol{v_1} \ \ \Vcol{v_2}\,]
\left[\begin{smallmatrix} 12 & -6\\ 0 & 6 \end{smallmatrix}\right]^{-1}$. No element value is
needed.
\end{examplebox}

\section{Th\'evenin and Norton equivalent circuits}

With independent sources inside, a one-port is no longer one resistor: it needs one more
element, a source. A load (the element connected to the port, e.g.\ a lamp or a speaker) connected to a large
circuit can then be studied with a two-element
circuit, solved once.

\begin{definitionbox}{Th\'evenin's theorem}
\begin{minipage}[c]{0.44\linewidth}
Let $N$ be an LTI resistive one-port, possibly with independent sources, such that
\begin{itemize}[leftmargin=2.4em, itemsep=1pt]
  \item[(A1)] every dependent source in $N$ has its control variable in $N$;
  \item[(A2)] the port is both voltage- and current-controlled.
\end{itemize}
Then there are $\vth$ and $\Rth$ with $N$ equivalent to the circuit on the right:
\[
  \Vcol{v} = \Rth\,\Icol{i} + \vth .
\]
\end{minipage}\hfill
\begin{minipage}[c]{0.54\linewidth}
\centering
\begin{tikzpicture}[scale=0.85]
  \oneport{$N$}
  \node[font=\large] at (4.0,1) {$\equiv$};
  \begin{scope}[shift={(5.6,0)}]
    \thev{$\vth$}{$\Rth$}
    \portiv{2.4}
  \end{scope}
\end{tikzpicture}
\end{minipage}
\end{definitionbox}

\begin{definitionbox}{Norton's theorem}
\begin{minipage}[c]{0.44\linewidth}
Under the same conditions (A1), (A2), there are $\iN$ and $\RN$ with $N$ equivalent to the
circuit on the right:
\[
  \Vcol{v} = \RN\,\Icol{i} + \RN\,\iN .
\]
\end{minipage}\hfill
\begin{minipage}[c]{0.54\linewidth}
\centering
\begin{tikzpicture}[scale=0.85]
  \oneport{$N$}
  \node[font=\large] at (4.0,1) {$\equiv$};
  \begin{scope}[shift={(5.6,0)}]
    \nort{$\iN$}{$\RN$}
    \portiv{2.95}
  \end{scope}
\end{tikzpicture}
\end{minipage}
\end{definitionbox}

\begin{minipage}[c]{0.62\linewidth}
\textbf{Th\'evenin and Norton describe the same line.} Both characteristics are the same
straight line in the $i$--$v$ plane, so
\[
  \Rth = \RN \qquad\text{and}\qquad \vth = \Rth\,\iN .
\]
Moving between the two is the source transformation (previously). The line meets the $v$ axis
at $\vth$ and the $i$ axis at $-\iN$; its slope is $\Rth$.
\end{minipage}\hfill
\begin{minipage}[c]{0.34\linewidth}
\centering
\begin{tikzpicture}[scale=0.8]
  \draw[-{Stealth[length=2.2mm]}] (-2.4,0) -- (1.8,0) node[right] {$\Icol{i}$};
  \draw[-{Stealth[length=2.2mm]}] (0,-0.9) -- (0,2.6) node[above] {$\Vcol{v}$};
  \draw[very thick, ccur] (-2.2,-0.6) -- (1.4,2.4);
  \fill (0,1.233) circle (2pt) node[below right, font=\small] {$\vth$};
  \fill (-1.48,0) circle (2pt) node[below right, font=\small] {$-\iN$};
  \node[font=\small, text=ccur] at (1.65,1.15) {slope $\Rth$};
\end{tikzpicture}
\end{minipage}

\section{Computing $\vth$, $\iN$ and $\Rth$}

\begin{center}
\begin{tikzpicture}[xscale=0.7, yscale=0.9]
  \Nport{0}
  \begin{scope}[shift={(6.0,0)}] \Nport{1} \end{scope}
  \begin{scope}[shift={(12.4,0)}] \Nport{2} \end{scope}
\end{tikzpicture}
\end{center}

\begin{enumerate}\itemsep2pt
  \item \textbf{Open circuit.} With the port open, $\Icol{i} = 0$ and the Th\'evenin
    characteristic gives $\Vcol{v} = \vth$: \ $\vth = \voc$.
  \item \textbf{Short circuit.} With the port shorted, $\Vcol{v} = 0$, so $\Icol{i} = -\iN$.
    The current through the short, from a to b, is $\isc = -\Icol{i}$: \ $\iN = \isc$.
  \item \textbf{$\boldsymbol{\Rth}$.} Once $\vth$ and $\iN$ are known, $\Rth = \vth/\iN =
    \voc/\isc$. Another way: kill all independent sources in $N$ (voltage source: short,
    current source: open) and compute the input resistance, $\Rth = \Rin$. Killing the
    independent sources sets $\vth = 0$ and $\iN = 0$; only $\Rth$ is left. The second way also
    works when $\voc = \isc = 0$.
\end{enumerate}

\section{Take-home: the $\boldsymbol{\Delta}$--Y transform}

Three resistors may form a $\Delta$ or a Y between three terminals A, B, C. In the $\Delta$,
$R_A$ is opposite A (between B and C), $R_B$ is opposite B and $R_C$ is opposite C. In the Y,
$R_1$, $R_2$, $R_3$ are at A, B, C.

\begin{center}
\begin{tikzpicture}[scale=0.8]
  \Dlt{0}
  \node[font=\Large] at (4.4,1.1) {$\equiv$};
  \begin{scope}[shift={(5.8,0)}]
    \Wye{0}
  \end{scope}
\end{tikzpicture}
\end{center}

\textbf{$\boldsymbol{\Delta}$ to Y.} Given $R_A$, $R_B$, $R_C$, find $R_1$, $R_2$, $R_3$. If the
two are equivalent, the resistance between any two terminals, with the third one open, is the
same. Between A and B, the $\Delta$ is $R_C$ in parallel with $R_A + R_B$, and the Y is
$R_1 + R_2$:
\[
  \frac{R_C(R_A + R_B)}{\SD} = R_1 + R_2 \ \ (1), \qquad
  \frac{R_A(R_B + R_C)}{\SD} = R_2 + R_3 \ \ (2),
\]
\[
  \frac{R_B(R_C + R_A)}{\SD} = R_3 + R_1 \ \ (3).
\]
Half of the sum of (1), (2) and (3) gives
\[
  R_1 + R_2 + R_3 = \frac{R_A R_B + R_B R_C + R_C R_A}{\SD} \qquad (4),
\]
and (4)$-$(2), (4)$-$(3), (4)$-$(1) give the result.

\begin{ideabox}[$\Delta$ to Y]
\[
  \Icol{R_1} = \frac{R_B R_C}{\SD}, \qquad
  \Icol{R_2} = \frac{R_C R_A}{\SD}, \qquad
  \Icol{R_3} = \frac{R_A R_B}{\SD}
\]
The Y resistor at a terminal is the product of the two $\Delta$ resistors at that terminal,
divided by the sum of the $\Delta$ resistors.
\end{ideabox}

\textbf{Y to $\boldsymbol{\Delta}$.} Given $R_1$, $R_2$, $R_3$, find $R_A$, $R_B$, $R_C$. From
the result above,
\[
  S = R_1R_2 + R_2R_3 + R_3R_1 = \frac{R_A R_B R_C\,(\SD)}{(\SD)^2} = \frac{R_A R_B R_C}{\SD},
\]
\[
  R_A R_1 = R_B R_2 = R_C R_3 = \frac{R_A R_B R_C}{\SD} = S .
\]

\begin{ideabox}[Y to $\Delta$]
\[
  \textcolor{metured}{R_A} = \frac{S}{R_1}, \qquad
  \textcolor{metured}{R_B} = \frac{S}{R_2}, \qquad
  \textcolor{metured}{R_C} = \frac{S}{R_3}, \qquad
  S = R_1R_2 + R_2R_3 + R_3R_1
\]
A $\Delta$ resistor is the sum of the products of the Y resistors, divided by the Y resistor
opposite to it.
\end{ideabox}

\begin{minipage}[c]{0.56\linewidth}
\textbf{Equal resistors.} A $\Delta$ of three equal resistors $R$ is equivalent to a Y of three
resistors $R \cdot R/(3R) = R/3$. Conversely, a Y of three resistors $R$ is equivalent to a
$\Delta$ of three resistors $3R$.

\smallskip
\textbf{Exercise.} Write both transforms with conductances, $G_\square = 1/R_\square$.
\emph{Answer:} Y to $\Delta$ becomes $G_A = G_2G_3/(G_1 + G_2 + G_3)$, and so on; it has the
form of $\Delta$ to Y with resistances.
\end{minipage}\hfill
